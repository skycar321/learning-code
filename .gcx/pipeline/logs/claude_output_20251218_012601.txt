```python
from typing import Union

def calculator(
    a: Union[int, float],
    b: Union[int, float],
    operation: str
) -> Union[int, float]:
    """
    두 숫자에 대한 기본 산술 연산을 수행하는 계산기 함수
    
    Args:
        a: 첫 번째 숫자
        b: 두 번째 숫자
        operation: 연산 종류 ('+', '-', '*', '/', '//', '%', '**')
    
    Returns:
        연산 결과값
    
    Raises:
        ValueError: 지원하지 않는 연산자인 경우
        ZeroDivisionError: 0으로 나누려고 할 때
    
    Examples:
        >>> calculator(10, 5, '+')
        15
        >>> calculator(10, 3, '//')
        3
        >>> calculator(2, 8, '**')
        256
    """
    if operation == '+':
        return a + b
    elif operation == '-':
        return a - b
    elif operation == '*':
        return a * b
    elif operation == '/':
        if b == 0:
            raise ZeroDivisionError("0으로 나눌 수 없습니다")
        return a / b
    elif operation == '//':
        if b == 0:
            raise ZeroDivisionError("0으로 나눌 수 없습니다")
        return a // b
    elif operation == '%':
        if b == 0:
            raise ZeroDivisionError("0으로 나눌 수 없습니다")
        return a % b
    elif operation == '**':
        return a ** b
    else:
        raise ValueError(f"지원하지 않는 연산자입니다: {operation}")


# 사용 예시
if __name__ == "__main__":
    # 기본 연산 테스트
    print(f"10 + 5 = {calculator(10, 5, '+')}")      # 15
    print(f"10 - 5 = {calculator(10, 5, '-')}")      # 5
    print(f"10 * 5 = {calculator(10, 5, '*')}")      # 50
    print(f"10 / 5 = {calculator(10, 5, '/')}")      # 2.0
    print(f"10 // 3 = {calculator(10, 3, '//')}")    # 3
    print(f"10 % 3 = {calculator(10, 3, '%')}")      # 1
    print(f"2 ** 8 = {calculator(2, 8, '**')}")      # 256
    
    # 실수 연산
    print(f"3.14 * 2 = {calculator(3.14, 2, '*')}")  # 6.28
    
    # 예외 처리 테스트
    try:
        calculator(10, 0, '/')
    except ZeroDivisionError as e:
        print(f"에러: {e}")
    
    try:
        calculator(10, 5, '^')
    except ValueError as e:
        print(f"에러: {e}")
```
